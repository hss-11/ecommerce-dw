            FROM v_order_item WHERE order_status = 'delivered'
            UNION ALL

            FROM v_order_item WHERE order_status = 'canceled'
            UNION ALL

            FROM v_order_item WHERE order_status NOT IN ('delivered', 'canceled', 'unavailable')
